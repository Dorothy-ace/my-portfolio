    def calculate_growth(initial_value, rate, steps):
    """Calculates step-by-step exponential growth."""
    results = []
    for t in range(steps):
        val = initial_value * ((1 + rate) ** t)
        results.append(round(val, 2))
    return results

print(calculate_growth(100, 0.05, 5))
